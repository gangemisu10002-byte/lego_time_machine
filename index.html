조명 연출 시스템

각 장면별로 다른 색온도의 LED를 사용하여 시간대와 분위기를 시각적으로 구분합니다.

Warm White (현재) Cool White (미래) Amber (적용)
📝 아두이노 핵심 코드 
// 타임머신 장면 전환 코드
#include <Servo.h>

Servo timeMachine;
int currentStage = 0;
const int buttonPin = 2;
const int ledPins[3] = {9, 10, 11}; // R, G, B

void setup() {
  timeMachine.attach(3);
  pinMode(buttonPin, INPUT_PULLUP);
  for(int i=0; i<3; i++) pinMode(ledPins[i], OUTPUT);
  Serial.begin(9600);
}

void loop() {
  if(digitalRead(buttonPin) == LOW) {
    nextStage();
    delay(300); // 디바운싱
  }
}

void nextStage() {
  currentStage = (currentStage + 1) % 3;
  
  // 서보 모터 회전 (0도, 120도, 240도)
  int angle = currentStage * 120;
  timeMachine.write(angle);
  
  // LED 색상 변경
  switch(currentStage) {
    case 0: setLED(255, 200, 100); break; // 따뜻한 백색
    case 1: setLED(100, 200, 255); break; // 차가운 청백색
    case 2: setLED(255, 150, 100); break; // 주황색
  }
  
  Serial.print("Stage: ");
  Serial.println(currentStage + 1);
}

void setLED(int r, int g, int b) {
  analogWrite(ledPins[0], r);
  analogWrite(ledPins[1], g);
  analogWrite(ledPins[2], b);
}
